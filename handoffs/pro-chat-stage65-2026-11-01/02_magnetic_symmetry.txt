    simp [hz]

def defectE (c : Coefficient) (S T : ℝ) (u : VectorL2) (i : Fin 3) (f : Test) (z : ℂ) : ℂ :=
  electricPair c z S T u (next i) (testPartial (previous i) f) -
    electricPair c z S T u (previous i) (testPartial (next i) f) - I*z*magneticPair c z S T u i f

def defectH (c : Coefficient) (S T : ℝ) (u : VectorL2) (i : Fin 3) (f : Test) (z : ℂ) : ℂ :=
  magneticPair c z S T u (next i) (testPartial (previous i) f) -
    magneticPair c z S T u (previous i) (testPartial (next i) f) + I*z*
      (electricPair c z S T u i f + ∫ x, DynamicVolume.source c u i x * f.val x)

theorem defectE_differentiable (c : Coefficient) (S T : ℝ) (hST : 0 ≤ T+S)
    (u : VectorL2) (i : Fin 3) (f : Test) : Differentiable ℂ (defectE c S T u i f) := by
  unfold defectE
  exact ((electricPair_differentiable c S T hST u _ _).sub (electricPair_differentiable c S T hST u _ _)).sub
    (((differentiable_const I).mul differentiable_id).mul (magneticPair_differentiable c S T u i f))

theorem defectH_differentiable (c : Coefficient) (S T : ℝ) (hST : 0 ≤ T+S)
    (u : VectorL2) (i : Fin 3) (f : Test) : Differentiable ℂ (defectH c S T u i f) := by
  unfold defectH
  exact ((magneticPair_differentiable c S T u _ _).sub (magneticPair_differentiable c S T u _ _)).add
    (((differentiable_const I).mul differentiable_id).mul ((electricPair_differentiable c S T hST u i f).add (differentiable_const _)))

theorem real_zero (c : Coefficient) (S T : ℝ) (u : VectorL2)
    (hc : ∀ x : Space, x ∉ closedBall 0 S → c.value (VectorFourier.coordinateMap x) = 0)
    (i : Fin 3) (f : Test) (hf : tsupport (f.val : Space → ℂ) ⊆ closedBall 0 T) (k : ℝ) :
    defectE c S T u i f k = 0 ∧ defectH c S T u i f k = 0 := by
  have hs (j : Fin 3) : tsupport ((testPartial j f).val : Space → ℂ) ⊆ closedBall 0 T :=
    (partial_support j f).trans hf
  have hE := MagneticResponse.electric_curl_pairing k (StaticCoefficient.sqrtMultiplier c u)
    (DynamicVolume.source c u) (DynamicVolume.source_integrable c u S hc)
    (MaterialField.source_compact c u S hc) (StaticNewton.material_source_ae c u) i f
  have hH := MagneticResponse.magnetic_curl_pairing k (StaticCoefficient.sqrtMultiplier c u)
    (DynamicVolume.source c u) (DynamicVolume.source_integrable c u S hc)
    (MaterialField.source_compact c u S hc) (StaticNewton.material_source_ae c u) i f
  have hmag : ComplexMagneticLocal.magnetic (k : ℂ) (DynamicVolume.source c u) =
      MagneticResponse.magnetic k (DynamicVolume.source c u) := rfl
  constructor
  · unfold defectE
    rw [electricPair_real c k S T u hc _ _ (hs _), electricPair_real c k S T u hc _ _ (hs _),
      magneticPair_actual c k S T u hc i f hf]
    rw [hmag]
    dsimp only [MaterialField.scattered]
    linear_combination hE
  · unfold defectH
    rw [magneticPair_actual c k S T u hc _ _ (hs _), magneticPair_actual c k S T u hc _ _ (hs _),
      electricPair_real c k S T u hc i f hf]
    rw [hmag]
    dsimp only [MaterialField.scattered]
    linear_combination hH

theorem analytic_zero_of_real (F : ℂ → ℂ) (hF : Differentiable ℂ F)
    (hreal : ∀ k : ℝ, F k = 0) (z : ℂ) : F z = 0 := by
  have hclosure : (1 : ℂ) ∈ closure ({w | F w = (0 : ℂ)} \ {(1 : ℂ)}) := by
    apply Metric.mem_closure_iff.mpr
    intro ε hε
    let r : ℝ := 1 + ε/2
    have hr1 : (r : ℂ) ≠ 1 := by
      have h : r ≠ 1 := by dsimp [r]; linarith
      exact_mod_cast h
    refine ⟨(r : ℂ), ⟨hreal r, hr1⟩, ?_⟩
    have hsub : (r : ℂ)-1 = ((ε/2 : ℝ) : ℂ) := by dsimp [r]; push_cast; ring
    rw [dist_comm, dist_eq_norm, hsub, Complex.norm_real, Real.norm_eq_abs, abs_of_pos (half_pos hε)]
    exact half_lt_self hε
  exact (show AnalyticOnNhd ℂ F Set.univ from fun w _ => hF.analyticAt w).eqOn_of_preconnected_of_mem_closure
    (fun w _ => analyticAt_const) isPreconnected_univ (mem_univ 1) hclosure (mem_univ z)

theorem complex_zero (c : Coefficient) (S T : ℝ) (hST : 0 ≤ T+S) (u : VectorL2)
    (hc : ∀ x : Space, x ∉ closedBall 0 S → c.value (VectorFourier.coordinateMap x) = 0)
    (i : Fin 3) (f : Test) (hf : tsupport (f.val : Space → ℂ) ⊆ closedBall 0 T) (z : ℂ) :
    defectE c S T u i f z = 0 ∧ defectH c S T u i f z = 0 := by
  constructor
  · exact analytic_zero_of_real _ (defectE_differentiable c S T hST u i f)
      (fun k => (real_zero c S T u hc i f hf k).1) z
  · exact analytic_zero_of_real _ (defectH_differentiable c S T hST u i f)
      (fun k => (real_zero c S T u hc i f hf k).2) z

theorem actual_maxwell (c : Coefficient) (S T : ℝ) (hST : 0 ≤ T+S) (u : VectorL2)
    (hc : ∀ x : Space, x ∉ closedBall 0 S → c.value (VectorFourier.coordinateMap x) = 0)
    (i : Fin 3) (f : Test) (hf : tsupport (f.val : Space → ℂ) ⊆ closedBall 0 T) (z : ℂ) :
    ((∫ x, ComplexExteriorResponse.response c z S T u (VectorFourier.coordinateMap x) (next i) * (testPartial (previous i) f).val x) -
      (∫ x, ComplexExteriorResponse.response c z S T u (VectorFourier.coordinateMap x) (previous i) * (testPartial (next i) f).val x) =
        I*z * (∫ x, ComplexMagneticLocal.magnetic z (DynamicVolume.source c u) x i * f.val x)) ∧
    ((∫ x, ComplexMagneticLocal.magnetic z (DynamicVolume.source c u) x (next i) * (testPartial (previous i) f).val x) -
      (∫ x, ComplexMagneticLocal.magnetic z (DynamicVolume.source c u) x (previous i) * (testPartial (next i) f).val x) =
        -(I*z) * ((∫ x, ComplexExteriorResponse.response c z S T u (VectorFourier.coordinateMap x) i * f.val x) +
          (∫ x, DynamicVolume.source c u i x * f.val x))) := by
  have hs (j : Fin 3) : tsupport ((testPartial j f).val : Space → ℂ) ⊆ closedBall 0 T :=
    (partial_support j f).trans hf
  obtain ⟨he,hh⟩ := complex_zero c S T hST u hc i f hf z
  unfold defectE at he
  unfold defectH at hh
  simp only [electricPair, testPair_integral] at he hh
  rw [magneticPair_actual c z S T u hc i f hf] at he
  rw [magneticPair_actual c z S T u hc _ _ (hs _), magneticPair_actual c z S T u hc _ _ (hs _)] at hh
  constructor
  · linear_combination he
  · linear_combination hh

    ComplexConvolutionAnalytic.volumeBlock (coefficient n hn) z 1 u ∈ electricSector := by
  refine ⟨ModalAxialVolume.axial_invariant n hn z u hu.1, ?_⟩
  have hf : action u = u := by
    have h := hu.2
    change (action - 1) u = 0 at h
    simpa using sub_eq_zero.mp h
  change (action - 1) (ComplexConvolutionAnalytic.volumeBlock (coefficient n hn) z 1 u) = 0
  have h := congrFun (congrArg DFunLike.coe (complex_comm n hn z)) u
  simp only [left, right, ContinuousLinearMap.comp_apply, hf] at h
  simpa using sub_eq_zero.mpr h

/-! # The physical Helmholtz kernel at complex frequency

The scalar kernel, its gradient and Hessian retain their actual real spatial
Frechet derivative meaning for every complex frequency. Restriction to real
frequency agrees exactly with the existing physical kernels.
-/

noncomputable section

def radial (k : ℂ) (r : ℝ) : ℂ := Complex.exp (Complex.I * k * r) / (r : ℂ)

def radialD (k : ℂ) (r : ℝ) : ℂ :=
  Complex.exp (Complex.I * k * r) * (Complex.I * k / r - 1 / (r : ℂ)^2)

def radialDD (k : ℂ) (r : ℝ) : ℂ :=
  Complex.exp (Complex.I * k * r) *
    ((Complex.I * k)^2 / r - 2 * (Complex.I * k) / (r : ℂ)^2 + 2 / (r : ℂ)^3)

theorem radial_hasDerivAt (k : ℂ) (r : ℝ) (hr : r ≠ 0) :
    HasDerivAt (radial k) (radialD k r) r := by
  have hc := complex_radius_hasDerivAt r
  have he := (hc.const_mul (Complex.I * k)).cexp
  have hrc : (r : ℂ) ≠ 0 := by exact_mod_cast hr
  convert he.div hc hrc using 1
  dsimp [radialD]
  field_simp [hrc]
  ring

theorem radialD_hasDerivAt (k : ℂ) (r : ℝ) (hr : r ≠ 0) :
    HasDerivAt (radialD k) (radialDD k r) r := by
  have hc := complex_radius_hasDerivAt r
  have he := (hc.const_mul (Complex.I * k)).cexp
  have hrc : (r : ℂ) ≠ 0 := by exact_mod_cast hr
  have h1 := (hasDerivAt_const r (Complex.I * k)).div hc hrc
  have hsq : HasDerivAt (fun t : ℝ => (t : ℂ)^2) (2 * (r : ℂ)) r := by
    convert hc.mul hc using 1 <;> ring
  have h2 := (hasDerivAt_const r (1 : ℂ)).div hsq (pow_ne_zero 2 hrc)
  convert he.mul (h1.sub h2) using 1
  dsimp [radialDD]
  field_simp [hrc]
  ring

theorem radial_helmholtz (k : ℂ) (r : ℝ) (hr : r ≠ 0) :
    radialDD k r + 2 / (r : ℂ) * radialD k r + k^2 * radial k r = 0 := by
  have hrc : (r : ℂ) ≠ 0 := by exact_mod_cast hr
  unfold radialDD radialD radial
  field_simp [hrc]
  ring_nf
  norm_num [Complex.I_sq]

def phi (k : ℂ) (x : Space) : ℂ := radial k ‖x‖ / (4 * Real.pi)

def phiGradient (k : ℂ) (x : Space) (j : Fin 3) : ℂ :=
  radialD k ‖x‖ * (x j / ‖x‖ : ℝ) / (4 * Real.pi)

theorem phi_partial (k : ℂ) (x : Space) (hx : x ≠ 0) (j : Fin 3) :
    dcoord (phi k) x j = phiGradient k x j := by
  have hr := norm_ne_zero_iff.mpr hx
  have hd := radial_hasDerivAt k ‖x‖ hr
  have hcomp := hd.hasFDerivAt.comp x (differentiableAt_id.norm ℝ hx).hasFDerivAt
  have hdiff : DifferentiableAt ℝ (fun y : Space => radial k ‖y‖) x :=
    hcomp.differentiableAt
  have hp := radial_composition_partial (radial k) (radialD k ‖x‖) x hx hd j
  have heq : phi k = fun y => (1 / (4 * Real.pi) : ℂ) * radial k ‖y‖ := by
    funext y
    simp [phi, div_eq_mul_inv, mul_comm]
  rw [heq, partial_const_mul hdiff, hp]
  simp [phiGradient, div_eq_mul_inv, mul_comm]

def phiHessian (k : ℂ) (x : Space) (i j : Fin 3) : ℂ :=
  (radialDD k ‖x‖ * (x i : ℂ) * (x j : ℂ) / (‖x‖ : ℂ)^2 +
    radialD k ‖x‖ * ((if i = j then 1 else 0 : ℂ) / (‖x‖ : ℂ) -
      (x i : ℂ) * (x j : ℂ) / (‖x‖ : ℂ)^3)) / (4 * Real.pi)

theorem phiGradient_partial (k : ℂ) (x : Space) (hx : x ≠ 0) (i j : Fin 3) :
    dcoord (fun y => phiGradient k y j) x i = phiHessian k x i j := by
  have hr : ‖x‖ ≠ 0 := norm_ne_zero_iff.mpr hx
  have hrc : (‖x‖ : ℂ) ≠ 0 := by exact_mod_cast hr
  have hd := radialD_hasDerivAt k ‖x‖ hr
  have hdiff : DifferentiableAt ℝ (fun y : Space => radialD k ‖y‖) x :=
    (hd.hasFDerivAt.comp x (differentiableAt_id.norm ℝ hx).hasFDerivAt).differentiableAt
  have hu := unitCoordinate_differentiableAt j x hx
  have heq : (fun y => phiGradient k y j) =
      fun y => (1 / (4 * Real.pi) : ℂ) * (radialD k ‖y‖ * unitCoordinate j y) := by
    funext y
    simp [phiGradient, unitCoordinate, div_eq_mul_inv, mul_comm, mul_left_comm]
  rw [heq, partial_const_mul (hdiff.mul hu), partial_mul hdiff hu,
    unitCoordinate_partial i j x hx, radial_composition_partial _ _ x hx hd i]
  unfold unitCoordinate phiHessian
  push_cast
  field_simp [hrc]
  ring

theorem phi_second_partial (k : ℂ) (x : Space) (hx : x ≠ 0) (i j : Fin 3) :
    dcoord (fun y => dcoord (phi k) y j) x i = phiHessian k x i j := by
  have he : (fun y => dcoord (phi k) y j) =ᶠ[nhds x] (fun y => phiGradient k y j) := by
    filter_upwards [eventually_ne_nhds hx] with y hy
    exact phi_partial k y hy j
  change (fderiv ℝ (fun y => dcoord (phi k) y j) x) (basis i) = _
  rw [he.fderiv_eq]
  exact phiGradient_partial k x hx i j

theorem phiHessian_trace (k : ℂ) (x : Space) (hx : x ≠ 0) :
    phiHessian k x 0 0 + phiHessian k x 1 1 + phiHessian k x 2 2 =
      (radialDD k ‖x‖ + 2 / (‖x‖ : ℂ) * radialD k ‖x‖) / (4 * Real.pi) := by
  have hr : (‖x‖ : ℂ) ≠ 0 := by exact_mod_cast norm_ne_zero_iff.mpr hx
  have hs := coordinate_squares x
  calc
    _ = (radialDD k ‖x‖ * ((x 0 : ℂ)^2 + (x 1 : ℂ)^2 + (x 2 : ℂ)^2) /
          (‖x‖ : ℂ)^2 + radialD k ‖x‖ * (3 / (‖x‖ : ℂ) -
          ((x 0 : ℂ)^2 + (x 1 : ℂ)^2 + (x 2 : ℂ)^2) / (‖x‖ : ℂ)^3)) /
          (4 * Real.pi) := by simp only [phiHessian, ite_true]; ring
    _ = _ := by
      rw [hs]
      congr 1
      field_simp [hr]
      ring

theorem phi_helmholtz (k : ℂ) (x : Space) (hx : x ≠ 0) :
    divergence (gradient (phi k)) x + k^2 * phi k x = 0 := by
  have hr : ‖x‖ ≠ 0 := norm_ne_zero_iff.mpr hx
  have ho := radial_helmholtz k ‖x‖ hr
  unfold divergence pd gradient
  rw [phi_second_partial k x hx, phi_second_partial k x hx, phi_second_partial k x hx,
    phiHessian_trace k x hx]
  unfold phi
  convert congrArg (fun z : ℂ => z / (4 * Real.pi)) ho using 1 <;> ring

theorem phi_contDiffAt (k : ℂ) (x : Space) (hx : x ≠ 0) :
    ContDiffAt ℝ ⊤ (phi k) x := by
  have hn : ContDiffAt ℝ ⊤ (fun y : Space => ‖y‖) x := contDiffAt_norm ℝ hx
  have hc := Complex.ofRealCLM.contDiff.contDiffAt.comp x hn
  have hr : (‖x‖ : ℂ) ≠ 0 := by exact_mod_cast norm_ne_zero_iff.mpr hx
  have hh := ((((contDiffAt_const (c := Complex.I * k)).mul hc).cexp).mul (hc.inv hr)).mul
    (contDiffAt_const (c := (4 * (Real.pi : ℂ))⁻¹))
  change ContDiffAt ℝ ⊤ (fun y : Space => phi k y) x
  simpa [phi, radial, div_eq_mul_inv, Function.comp_def] using hh
