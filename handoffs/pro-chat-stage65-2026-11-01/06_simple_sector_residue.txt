theorem scalarPair_smul_left (a : ℂ) (F G : ScalarL2) :
    scalarPair (a • F) G = a * scalarPair F G := by
  unfold scalarPair
  calc
    _ = ∫ x, a * (F x * G x) := by
      apply integral_congr_ae
      filter_upwards [Lp.coeFn_smul a F] with x hx
      simp only [hx, Pi.smul_apply, smul_eq_mul, mul_assoc]
    _ = _ := integral_const_mul a _

theorem vectorPair_smul_left (a : ℂ) (F G : VectorL2) :
    vectorPair (a • F) G = a * vectorPair F G := by
  simp only [vectorPair, map_smul, scalarPair_smul_left, Finset.mul_sum]

def scalarPairFunctional (G : ScalarL2) : ScalarL2 →L[ℂ] ℂ where
  toFun F := scalarPair F G
  map_add' := fun F H => scalarPair_add_left F H G
  map_smul' := fun a F => scalarPair_smul_left a F G
  cont := by
    exact scalarPair_continuous.comp (f := fun F : ScalarL2 => (F, G))
      ((continuous_id : Continuous (fun F : ScalarL2 => F)).prodMk
        (continuous_const : Continuous (fun _ : ScalarL2 => G)))

def scalarReciprocityDefect (F G : ScalarL2) : (ScalarL2 →L[ℂ] ScalarL2) →L[ℂ] ℂ :=
  (scalarPairFunctional G).comp (ContinuousLinearMap.apply ℂ ScalarL2 F) -
    (scalarPairFunctional F).comp (ContinuousLinearMap.apply ℂ ScalarL2 G)

theorem scalarReciprocityDefect_apply (A : ScalarL2 →L[ℂ] ScalarL2) (F G : ScalarL2) :
    scalarReciprocityDefect F G A = scalarPair (A F) G - scalarPair F (A G) := by
  change scalarPair (A F) G - scalarPair (A G) F = _
  rw [scalarPair_comm (A G) F]

def pairFunctional (G : VectorL2) : VectorL2 →L[ℂ] ℂ where
  toFun F := vectorPair F G
  map_add' := fun F H => vectorPair_add_left F H G
  map_smul' := fun a F => vectorPair_smul_left a F G
  cont := by

        (continuous_const : Continuous (fun _ : VectorL2 => G)))

theorem preconditioned_eq (c : Coefficient) (k R : ℝ) :
    Ring.inverse (StaticNewton.volumeBlock c) * DynamicVolume.volumeBlock c k R =
      CompactPerturbation.defect (compactPart c k R) := by
  change Ring.inverse (StaticNewton.volumeBlock c) *
    (StaticNewton.volumeBlock c - DynamicVolume.weightedRemainder c k R) =
      1 - Ring.inverse (StaticNewton.volumeBlock c) * DynamicVolume.weightedRemainder c k R
  rw [mul_sub, Ring.inverse_mul_cancel _ (StaticNewton.volumeBlock_isUnit c)]

    DynamicVolume.volumeBlock c k R =
      StaticNewton.volumeBlock c * CompactPerturbation.defect (compactPart c k R) := by
  rw [← preconditioned_eq, ← mul_assoc,
    Ring.mul_inverse_cancel _ (StaticNewton.volumeBlock_isUnit c), one_mul]

    LinearMap.ker (DynamicVolume.volumeBlock c k R) =
      LinearMap.ker (CompactPerturbation.defect (compactPart c k R)) := by
  ext x
  change DynamicVolume.volumeBlock c k R x = 0 ↔
    CompactPerturbation.defect (compactPart c k R) x = 0
  rw [volumeBlock_factor]
  change StaticNewton.volumeBlock c
    (CompactPerturbation.defect (compactPart c k R) x) = 0 ↔ _
  exact (ContinuousLinearMap.isUnit_iff_bijective.mp (StaticNewton.volumeBlock_isUnit c)).1.eq_iff'

    IsClosed (Set.range (DynamicVolume.volumeBlock c k R)) := by
  have he : Set.range (DynamicVolume.volumeBlock c k R) =
      (fun y => (Ring.inverse (StaticNewton.volumeBlock c)) y) ⁻¹'
        Set.range (CompactPerturbation.defect (compactPart c k R)) := by
    ext y
    constructor
    · rintro ⟨x, rfl⟩
      refine ⟨x, ?_⟩
      exact (congrArg (fun L : VectorL2 →L[ℂ] VectorL2 => L x) (preconditioned_eq c k R)).symm

      refine ⟨x, ?_⟩
      rw [volumeBlock_factor]
      change StaticNewton.volumeBlock c
        (CompactPerturbation.defect (compactPart c k R) x) = y
      rw [hx]
      exact congrArg (fun L : VectorL2 →L[ℂ] VectorL2 => L y)
        (Ring.mul_inverse_cancel _ (StaticNewton.volumeBlock_isUnit c))
  rw [he]
  exact (CompactPerturbation.range_closed _ (compactPart_compact c k R hc)).preimage

  obtain ⟨C, hC, hb⟩ := CompactPerturbation.orthogonal_norm_bound _

  rw [kernel_eq] at hu

  exact ((Ring.inverse (StaticNewton.volumeBlock c)).le_opNorm _).trans (by simpa using hn)

variable {𝕜 E V : Type*} [NontriviallyNormedField 𝕜]
  [NormedAddCommGroup E] [NormedSpace 𝕜 E]
  [NormedAddCommGroup V] [NormedSpace 𝕜 V]

def large (U : V →L[𝕜] E) (B : E →L[𝕜] V) : E →L[𝕜] E := 1-U.comp B

def small (U : V →L[𝕜] E) (B : E →L[𝕜] V) : V →L[𝕜] V := 1-B.comp U

def liftedInverse (U : V →L[𝕜] E) (B : E →L[𝕜] V) : E →L[𝕜] E :=
  1+U.comp ((Ring.inverse (small U B)).comp B)

theorem large_liftedInverse (U : V →L[𝕜] E) (B : E →L[𝕜] V)
    (hD : IsUnit (small U B)) : large U B * liftedInverse U B = 1 := by
  apply ContinuousLinearMap.ext
  intro x
  have h := congrArg (fun T : V →L[𝕜] V => T (B x))
    (Ring.mul_inverse_cancel (small U B) hD)
  change Ring.inverse (small U B) (B x)-B (U (Ring.inverse (small U B) (B x))) = B x at h
  have he : B x+B (U (Ring.inverse (small U B) (B x))) =
      Ring.inverse (small U B) (B x) := (eq_add_of_sub_eq h).symm
  change x+U (Ring.inverse (small U B) (B x))-
    U (B (x+U (Ring.inverse (small U B) (B x)))) = x
  rw [map_add, he]
  abel

theorem liftedInverse_large (U : V →L[𝕜] E) (B : E →L[𝕜] V)
    (hD : IsUnit (small U B)) : liftedInverse U B * large U B = 1 := by
  apply ContinuousLinearMap.ext
  intro x
  have h := congrArg (fun T : V →L[𝕜] V => T (B x))
    (Ring.inverse_mul_cancel (small U B) hD)
  change Ring.inverse (small U B) (B x-B (U (B x))) = B x at h
  change x-U (B x)+U (Ring.inverse (small U B) (B (x-U (B x)))) = x
  rw [map_sub, h]
  abel

theorem large_isUnit_of_small (U : V →L[𝕜] E) (B : E →L[𝕜] V)
    (hD : IsUnit (small U B)) : IsUnit (large U B) := by
  exact ⟨⟨large U B, liftedInverse U B, large_liftedInverse U B hD,
    liftedInverse_large U B hD⟩,rfl⟩

theorem large_isUnit_iff_small (U : V →L[𝕜] E) (B : E →L[𝕜] V) :
    IsUnit (large U B) ↔ IsUnit (small U B) :=
  ⟨fun h => large_isUnit_of_small B U h, large_isUnit_of_small U B⟩

theorem inverse_large (U : V →L[𝕜] E) (B : E →L[𝕜] V)
    (hD : IsUnit (small U B)) : Ring.inverse (large U B) = liftedInverse U B := by
  have hF := large_isUnit_of_small U B hD
  calc
    Ring.inverse (large U B) = Ring.inverse (large U B)*1 := by rw [mul_one]
    _ = Ring.inverse (large U B)*(large U B*liftedInverse U B) := by
      rw [large_liftedInverse U B hD]
    _ = liftedInverse U B := by
      rw [← mul_assoc, Ring.inverse_mul_cancel (large U B) hF, one_mul]

def schur (A : E →L[𝕜] E) (I : V →L[𝕜] E) (B : E →L[𝕜] V) : V →L[𝕜] V :=
  1-B.comp ((Ring.inverse A).comp I)

def woodbury (A : E →L[𝕜] E) (I : V →L[𝕜] E) (B : E →L[𝕜] V) : E →L[𝕜] E :=
  Ring.inverse A+(Ring.inverse A).comp
    (I.comp ((Ring.inverse (schur A I B)).comp (B.comp (Ring.inverse A))))

theorem factorization (A : E →L[𝕜] E) (hA : IsUnit A)
    (I : V →L[𝕜] E) (B : E →L[𝕜] V) :
    A-I.comp B = A*large ((Ring.inverse A).comp I) B := by
  apply ContinuousLinearMap.ext
  intro x
  have h := congrArg (fun T : E →L[𝕜] E => T (I (B x))) (Ring.mul_inverse_cancel A hA)
  simp only [ContinuousLinearMap.mul_apply, ContinuousLinearMap.one_apply] at h
  simp only [ContinuousLinearMap.sub_apply, ContinuousLinearMap.comp_apply,
    ContinuousLinearMap.mul_apply, large, ContinuousLinearMap.one_apply, map_sub, h]

/-- Invertibility of the Banach-space perturbation is exactly invertibility
of its auxiliary-space Schur operator. -/
theorem isUnit_iff_schur (A : E →L[𝕜] E) (hA : IsUnit A)
    (I : V →L[𝕜] E) (B : E →L[𝕜] V) :
    IsUnit (A-I.comp B) ↔ IsUnit (schur A I B) := by
  constructor
  · intro hF
    have hL : IsUnit (large ((Ring.inverse A).comp I) B) := by
      have h := hA.ringInverse.mul hF
      rw [factorization A hA I B, ← mul_assoc, Ring.inverse_mul_cancel A hA, one_mul] at h
      exact h
    exact (large_isUnit_iff_small ((Ring.inverse A).comp I) B).mp hL
  · intro hD
    rw [factorization A hA I B]
    exact hA.mul (large_isUnit_of_small ((Ring.inverse A).comp I) B hD)

theorem woodbury_eq_product (A : E →L[𝕜] E)
    (I : V →L[𝕜] E) (B : E →L[𝕜] V) :
    woodbury A I B = liftedInverse ((Ring.inverse A).comp I) B * Ring.inverse A := by
  apply ContinuousLinearMap.ext
  intro x
  simp only [woodbury, liftedInverse, ContinuousLinearMap.add_apply,
    ContinuousLinearMap.comp_apply, ContinuousLinearMap.mul_apply, ContinuousLinearMap.one_apply]
  rfl

theorem perturbation_woodbury (A : E →L[𝕜] E) (hA : IsUnit A)
    (I : V →L[𝕜] E) (B : E →L[𝕜] V) (hD : IsUnit (schur A I B)) :
    (A-I.comp B)*woodbury A I B = 1 := by
  rw [factorization A hA I B, woodbury_eq_product, mul_assoc,
    ← mul_assoc (large _ _) (liftedInverse _ _), large_liftedInverse _ _ hD,
    one_mul, Ring.mul_inverse_cancel A hA]

theorem woodbury_perturbation (A : E →L[𝕜] E) (hA : IsUnit A)
    (I : V →L[𝕜] E) (B : E →L[𝕜] V) (hD : IsUnit (schur A I B)) :
    woodbury A I B*(A-I.comp B) = 1 := by
  rw [factorization A hA I B, woodbury_eq_product,
    mul_assoc, ← mul_assoc (Ring.inverse A) A, Ring.inverse_mul_cancel A hA,
    one_mul, liftedInverse_large _ _ hD]

theorem inverse_woodbury (A : E →L[𝕜] E) (hA : IsUnit A)
    (I : V →L[𝕜] E) (B : E →L[𝕜] V) (hD : IsUnit (schur A I B)) :
    Ring.inverse (A-I.comp B) = woodbury A I B := by
  have hF := (isUnit_iff_schur A hA I B).mpr hD
  calc
    Ring.inverse (A-I.comp B) = Ring.inverse (A-I.comp B)*1 := by rw [mul_one]
    _ = Ring.inverse (A-I.comp B)*((A-I.comp B)*woodbury A I B) := by
      rw [perturbation_woodbury A hA I B hD]
    _ = woodbury A I B := by
      rw [← mul_assoc, Ring.inverse_mul_cancel _ hF, one_mul]

end OQ14.FiniteRankSchur

/-! Form of the residue of the inverse of an analytic operator family at a simple pole: if the
kernel at the pole is spanned by `u`, the covector `φ` kills the range, and `φ (A' u) ≠ 0`, then any
operator-norm residue limit equals `x ↦ (φ x / φ (A' u)) • u`. -/

variable {H : Type*} [NormedAddCommGroup H] [NormedSpace ℂ H]

theorem residue_kills_kernel (A : ℂ → H →L[ℂ] H) (A1 : H →L[ℂ] H) (z0 : ℂ) (hA : HasDerivAt A A1 z0)
    (hinv : ∀ᶠ z in 𝓝[≠] z0, IsUnit (A z)) (R : H →L[ℂ] H)
    (hR : Tendsto (fun z => (z-z0) • Ring.inverse (A z)) (𝓝[≠] z0) (𝓝 R)) :
    A z0 * R=0 := by
  have hc : Tendsto A (𝓝[≠] z0) (𝓝 (A z0)) := hA.continuousAt.tendsto.mono_left nhdsWithin_le_nhds
  have h1 : Tendsto (fun z => A z * ((z-z0) • Ring.inverse (A z))) (𝓝[≠] z0) (𝓝 (A z0 * R)) := hc.mul hR
  have h2 : Tendsto (fun z : ℂ => A z * ((z-z0) • Ring.inverse (A z))) (𝓝[≠] z0) (𝓝 0) := by
    have hz : Tendsto (fun z : ℂ => z-z0) (𝓝[≠] z0) (𝓝 0) := by
      have h0 : Tendsto (fun z : ℂ => z-z0) (𝓝 z0) (𝓝 0) := by
        have := (continuous_id.sub (continuous_const (y := z0))).tendsto z0
        simpa using this
      exact h0.mono_left nhdsWithin_le_nhds
    have h3 : Tendsto (fun z : ℂ => (z-z0) • (1 : H →L[ℂ] H)) (𝓝[≠] z0) (𝓝 0) := by
      simpa using hz.smul_const (1 : H →L[ℂ] H)
    refine h3.congr' ?_
    filter_upwards [hinv] with z hz
    rw [mul_smul_comm,Ring.mul_inverse_cancel _ hz]
  exact tendsto_nhds_unique h1 h2

theorem residue_form (A : ℂ → H →L[ℂ] H) (A1 : H →L[ℂ] H) (z0 : ℂ) (hA : HasDerivAt A A1 z0)
    (u : H) (φ : H →L[ℂ] ℂ)
    (hker : ∀ v, A z0 v=0 → ∃ a : ℂ,v=a • u)
    (hφ : ∀ v, φ (A z0 v)=0) (hd : φ (A1 u)≠0)
    (hinv : ∀ᶠ z in 𝓝[≠] z0, IsUnit (A z)) (R : H →L[ℂ] H)
    (hR : Tendsto (fun z => (z-z0) • Ring.inverse (A z)) (𝓝[≠] z0) (𝓝 R)) :
    ∀ x, R x=(φ x/φ (A1 u)) • u := by
  intro x
  have hk : A z0 (R x)=0 := by
    have := residue_kills_kernel A A1 z0 hA hinv R hR
    exact congrArg (fun T : H →L[ℂ] H => T x) this
  obtain ⟨a,ha⟩ := hker _ hk
  have hs : Tendsto (slope A z0) (𝓝[≠] z0) (𝓝 A1) := hasDerivAt_iff_tendsto_slope.mp hA
  have hlim : Tendsto (fun z => φ ((slope A z0 z * ((z-z0) • Ring.inverse (A z))) x)) (𝓝[≠] z0)
      (𝓝 (φ ((A1 * R) x))) := by
    have h := hs.mul hR
    exact φ.continuous.continuousAt.tendsto.comp
      ((ContinuousLinearMap.apply ℂ H x).continuous.continuousAt.tendsto.comp h)
  have heq : ∀ᶠ z in 𝓝[≠] z0, φ ((slope A z0 z * ((z-z0) • Ring.inverse (A z))) x)=φ x := by
    filter_upwards [hinv,self_mem_nhdsWithin] with z hz hne
    have hne' : z-z0≠0 := sub_ne_zero.mpr hne
    have h1 : (A z) (((z-z0) • Ring.inverse (A z)) x)=(z-z0) • x := by
      have := congrArg (fun T : H →L[ℂ] H => T x) (Ring.mul_inverse_cancel _ hz)
      simp only [ContinuousLinearMap.mul_apply,ContinuousLinearMap.one_apply] at this
      simp only [ContinuousLinearMap.smul_apply,map_smul,this]
    have h2 : slope A z0 z=(z-z0)⁻¹ • (A z-A z0) := by simp [slope_def_module]
    set y := ((z-z0) • Ring.inverse (A z)) x with hy
    have h3 : (slope A z0 z) y=(z-z0)⁻¹ • ((A z) y-(A z0) y) := by
      rw [h2]; simp
    have h4 : φ ((slope A z0 z) y)=φ x := by
      rw [h3,map_smul,map_sub,hφ,h1]
      simp [map_smul, hne']
    rw [ContinuousLinearMap.mul_apply]
    exact h4
  have hconst : Tendsto (fun z : ℂ => φ x) (𝓝[≠] z0) (𝓝 (φ x)) := tendsto_const_nhds

    tendsto_nhds_unique hlim (hconst.congr' (heq.mono fun z hz => hz.symm))
  rw [ContinuousLinearMap.mul_apply,ha,map_smul,map_smul,smul_eq_mul] at hxe
  rw [ha]
  congr 1
  field_simp
  exact hxe

theorem prescribed_frequency (n : ℝ) (hn : 1 < n)
    (ω : ℝ) (hω : 0 < ω) (Q : ℝ) (hQ : 0 ≤ Q) :
    ∃ s : ℝ, 0 < s ∧ ∃ z : ℂ, ∃ u : VectorFourier.VectorL2,
      z.re = ω ∧ z.im < 0 ∧ Q < z.re / (-2 * z.im) ∧
      (coefficient n hn s).bound = n ^ 2 - 1 ∧
      (∀ x : Maxwell.Space, x ∉ closedBall 0 (1 / s) →
        (coefficient n hn s).value (VectorFourier.coordinateMap x) = 0) ∧
      u ≠ 0 ∧ volumeBlock (coefficient n hn s) z (1 / s) u = 0 ∧
      pairFunctional u ≠ 0 ∧
      (∀ v, pairFunctional u (volumeBlock (coefficient n hn s) z (1 / s) v) = 0) ∧
      pairFunctional u (deriv (fun w => volumeBlock (coefficient n hn s) w (1 / s)) z u) = 1 ∧
      (∃ hm : MeromorphicAt
        (fun w => Ring.inverse (volumeBlock (coefficient n hn s) w (1 / s))) z,
        hm.order < 0) ∧
      (∀ᶠ w in 𝓝[≠] z, IsUnit (volumeBlock (coefficient n hn s) w (1 / s))) := by
  obtain ⟨k, _, hk, hi, hquality, hd, hsimple, _, _, _, _⟩ :=
    ModalSphereVolumePoles.high_quality_volume_poles n hn 0 Q hQ
  obtain ⟨f, a, _, hu0, hker0, _, _, hunit0⟩ :=
    ModalVolumeNormalization.exists_normalized_weighted_mode n hn k hd hsimple
  let u0 := a • ModalWeightedMode.weightedMode n hn k f
  let s := ω / k.val.re
  have hs : 0 < s := div_pos hω hk
  have hs0 : (s : ℂ) ≠ 0 := ofReal_ne_zero.mpr hs.ne'
  let z := (s : ℂ) * k.val
  let u := (s : ℂ) • DilationL2.dilate s u0
  have hre : z.re = s * k.val.re := by simp [z]
  have him : z.im = s * k.val.im := by simp [z]
  have htarget : z.re = ω := by rw [hre]; exact div_mul_cancel₀ ω hk.ne'
  have hnegative : z.im < 0 := by rw [him]; exact mul_neg_of_pos_of_neg hs hi
  have hratio : z.re / (-2 * z.im) = k.val.re / (-2 * k.val.im) := by
    rw [hre, him]
    have hi0 : k.val.im ≠ 0 := ne_of_lt hi
    field_simp
    ring
  have htransport := ComplexDilationVolume.nonzero_kernel_transport s 1 hs (by norm_num)
    (ModalWeightedMode.coefficient n hn) k.val u0 hu0 hker0
  have hu : u ≠ 0 := smul_ne_zero hs0 htransport.1
  have hker : volumeBlock (coefficient n hn s) z (1 / s) u = 0 := by
    change volumeBlock (FrequencyScalingMaxwell.scaleCoeff s (ModalWeightedMode.coefficient n hn))
      ((s : ℂ)*k.val) (1/s) ((s : ℂ) • DilationL2.dilate s u0) = 0
    rw [map_smul, htransport.2, smul_zero]
  have hunit : pairFunctional u
      (deriv (fun w => volumeBlock (coefficient n hn s) w (1 / s)) z u) = 1 :=
    DilationReciprocity.unit_pairing_transport s 1 hs (by norm_num)
      (ModalWeightedMode.coefficient n hn) k.val u0 hunit0
  have hleft (v : VectorFourier.VectorL2) :
      pairFunctional u (volumeBlock (coefficient n hn s) z (1 / s) v) = 0 := by
    change vectorPair (volumeBlock (coefficient n hn s) z (1 / s) v) u = 0
    rw [volumeBlock_reciprocity _ (1/s) (one_div_pos.mpr hs).le, hker, vectorPair_comm]
    exact (pairFunctional v).map_zero
  have hfunctional : pairFunctional u ≠ 0 := by
    intro hzero
    have h := hunit
    rw [hzero, ContinuousLinearMap.zero_apply] at h
    exact zero_ne_one h
  exact ⟨s, hs, z, u, htarget, hnegative, hratio ▸ hquality,
    coefficient_bound n hn s, coefficient_support n hn s hs, hu, hker, hfunctional,
    hleft, hunit, scaled_pole n hn s hs k.val k.property hd,
    VolumeAnalyticAlternative.volumeBlock_punctured_isUnit _ _
      (one_div_pos.mpr hs).le (coefficient_support n hn s hs) z⟩

end OQ14.ModalFixedFrequencyNormalization

\subsection{A compact analytic pencil and continuity under rounding}

Let $B$ be a ball containing all material changes, let
$X=L^2(B;\C^3)$, and extend its elements by zero when applying
whole-space operators.  Write $\chi=\varepsilon_r-1$.  With
$\Phi_k(x)=e^{ik|x|}/(4\pi|x|)$, put
\begin{equation}
 K(k)f=\left.(k^2I+\nabla\nabla)\Phi_k*f\right|_B,
 \qquad L_\chi(k)=I-K(k)\chi.
 \label{ms:pencil}
\end{equation}
Multiplication by $\chi$ is included in the last expression.
The static longitudinal Fourier projection, compressed to $B$, is
denoted by $P$.  It satisfies $P=P^*$ and $0\leq P\leq I$.
The distributional identity
\begin{equation}
 K(k)=-P+R(k),\qquad
 R(k)=k^2\Phi_k I+\nabla\nabla(\Phi_k-\Phi_0),\qquad R(0)=0
 \label{ms:compact-split}
\end{equation}
holds as an identity of bounded operators on $X$.

\begin{lemma}[PROVED: analytic Fredholm reduction with a fixed material bound]
\label{ms:fredholm}
For every measurable $0\leq\chi\leq C$ supported in $B$,
\begin{equation}
 L_\chi(k)=A_\chi F_\chi(k),\quad
 A_\chi=I+P\chi,\quad
 F_\chi(k)=I-A_\chi^{-1}R(k)\chi
 \label{ms:factorization}
\end{equation}
has $A_\chi$ invertible with $\|A_\chi^{-1}\|\leq1+C$.
The operator $R(k)$ is an entire Hilbert--Schmidt family, and
$F_\chi$ is an analytic identity-minus-compact pencil with
$F_\chi(0)=I$.  Its noninvertible values are isolated, of finite
algebraic multiplicity.  At $k\ne0$, its kernel is exactly the
restriction to $B$ of the finite local-energy Maxwell resonances
for this material, with exterior field given by \eqref{ms:pencil}.

If $0\leq\chi_j\leq C$ and $\chi_j\to\chi$ almost everywhere
in this same ball, then $F_{\chi_j}(k)\to F_\chi(k)$ in operator
norm, locally uniformly in $k$, as do all $k$ derivatives.
\end{lemma}

\begin{proof}
The Fourier symbol of $\nabla\nabla\Phi_0$ is
$-\xi\xi^T/|\xi|^2$, including its distributional local term.
Taylor expansion of $e^{ikr}$ shows that each entry of the
remaining kernel in \eqref{ms:compact-split} is $O(1/r)$ as
$r\downarrow0$, uniformly for $k$ in compact subsets of $\C$.
The constant term $ik/(4\pi)$ in $\Phi_k-\Phi_0$ has zero Hessian.
The squared kernel is locally integrable in three dimensions;
on bounded $B\times B$ it is integrable.  Its parameter power
series and derivatives have the same local majorants on compact
parameter sets.  This proves the entire Hilbert--Schmidt assertion.

Put $s=\sqrt\chi$.  The positive selfadjoint operator $I+sPs$
has inverse of norm at most one, and direct multiplication gives
\begin{equation}
 A_\chi^{-1}=I-Ps(I+sPs)^{-1}s.
 \label{ms:static-inverse}
\end{equation}
This proves the stated bound and factorization.  Analytic Fredholm
theory applies, with the invertible value at zero fixing its
nontrivial alternative.  This application needs no small contrast,
smooth boundary, or uniform feature size.

For completeness, the volume representation is an exact Maxwell
equation.  In distributions,
$(\operatorname{curl}\operatorname{curl}-k^2)K(k)f=k^2f$.
Consequently a kernel vector extended by $E=K(k)(\chi E)$
satisfies $\operatorname{curl}\operatorname{curl}E
=k^2(1+\chi)E$.  Its curl is locally square integrable, by
the first-derivative Newton-potential estimate; equivalently this
follows from local cutoff testing.  Set $H=(ik)^{-1}\operatorname{curl}E$
in units with vacuum impedance one.  This gives the two first-order
Maxwell equations and the outgoing continued representation.
Conversely, subtracting this volume field from any such resonance
leaves an entire vacuum Maxwell field regular at the origin and
outgoing outside a ball.  In each vector spherical harmonic its
regular radial function and outgoing radial function have nonzero
Wronskian for $k\ne0$, so all coefficients vanish.  Thus the
restriction is a kernel vector.  A zero restriction gives the zero
global field.

Almost-everywhere convergence and the common bound give strong
convergence of the multiplication operators.  The resolvent identity
and \eqref{ms:static-inverse} give strong convergence of $A_{\chi_j}^{-1}$
with a uniform bound.  Dominated convergence of the integral kernels
gives $R(k)\chi_j\to R(k)\chi$ in Hilbert--Schmidt norm.
A uniformly bounded strongly convergent operator sequence converges
in norm after composition on the right with a fixed compact operator:
approximate that compact operator by finite-rank operators.
Apply this observation to $A_{\chi_j}^{-1}R(k)\chi$ and add the
Hilbert--Schmidt difference.  Compactness in the parameter $k$ makes
the convergence locally uniform; Cauchy's formula gives the same
claim for derivatives.
\end{proof}

All these arguments also hold on a fixed closed symmetry sector
preserved by $P$, $R(k)$, and the material multiplication.  We use
the axisymmetric electric sector: $E$ has only meridional components
and $H$ only the azimuthal component in cylindrical coordinates.
Rotationally invariant scalar material preserves this sector.
It is closed in $L^2$ and orthogonal to the axisymmetric magnetic
sector, both for the Hermitian pairing and for the unconjugated
bilinear pairing.  At the coordinate axis these statements are
understood in the distributional and almost-everywhere sense.

\begin{lemma}[PROVED: persistence of a simple resonance]
\label{ms:continuation}
Suppose the restriction of $F_\chi$ to such a fixed sector has an
isolated characteristic value $k_*$ of algebraic multiplicity one.
An analytic $L^\infty$ material perturbation has a locally
holomorphic continued wavenumber $k_\eta$ and nonzero eigenfield
$E_\eta\in X$.  Their values at zero are $k_*$ and a prescribed
nonzero eigenfield.  If instead bounded nonnegative materials
$\chi_j$ converge almost everywhere to $\chi$, and preserve this
sector, they have unique nearby simple characteristic values
$k_j\to k_*$ and eigenfields $E_j\to E$ in $X$, after one fixed
nonzero linear functional fixes their scale.
\end{lemma}

\begin{proof}
Choose closed complements of the one-dimensional kernel and
cokernel of $F_\chi(k_*)$.  The block between those complements
is invertible, and remains so in a small operator-norm neighborhood.
Block elimination reduces the kernel equation to a single scalar
analytic function $f(k)$, multiplied by invertible analytic block
operators.  Algebraic multiplicity one is exactly
$f(k_*)=0$, $f'(k_*)\ne0$.
For an analytic material perturbation, $A_\chi^{-1}$ continues
analytically by a Neumann series near the base material, so every
block is jointly analytic.  The scalar analytic implicit-function
theorem supplies $k_\eta$; back substitution supplies $E_\eta$.
For almost-everywhere material convergence, Lemma~\ref{ms:fredholm}
gives uniform convergence of the blocks and scalar functions on a
small closed disk.  Rouch\'e's theorem gives exactly one zero,
counting multiplicity.  It tends to $k_*$ and is simple.  Back
substitution and a linear functional nonzero on $E$ give the stated
field convergence.  This reduction must use $F$ for the
almost-everywhere convergence claim: the leading multiplication
in $L$ need not converge in operator norm.  Multiplication by $A$
is analytic and invertible and does not alter a characteristic
value's multiplicity or its right kernel.
\end{proof}

The bilinear pairing in the remainder of this section has no
complex conjugation.  Reciprocity gives $K(k)^T=K(k)$ on $X$.
For a simple characteristic value in the selected sector, define
\begin{equation}
 d(E,k,\chi)=-\int_B \chi E\cdot K'(k)(\chi E)\,dx,
 \qquad \mathcal N(E)=-\frac{k}{2}d(E,k,\chi).
 \label{ms:residue-norm}
\end{equation}
The derivative kernel is a bounded compact operator; these are
finite integrals of locally square-integrable fields.  Their values
are independent of which containing ball is used.

\begin{lemma}[PROVED: nonzero residue norm and its stability]
\label{ms:normalization}
At a simple nonzero Maxwell pole in the selected sector,
$\mathcal N(E)\ne0$.  Scaling $E$ by $a\in\C$ scales
$\mathcal N$ by $a^2$.  The reciprocal electric Green operator for
$\operatorname{curl}\operatorname{curl}-k^2\varepsilon_r$,
restricted to this sector, has residue
\begin{equation}
 \operatorname*{Res}_{k=k_*}G(k;x,y)
       =-\frac{E(x)\otimes E(y)}{2k_*\mathcal N(E)}.
 \label{ms:green-residue}
\end{equation}
The norm is continuous along the almost-everywhere material and
eigenfield convergence of Lemma~\ref{ms:continuation}, and
holomorphic along its analytic branches.  A local nonzero analytic
choice of scale makes $\mathcal N=1$.
\end{lemma}

\begin{proof}
The functional $\ell_E(v)=\int_B\chi E\cdot v$ is nonzero:
if $\chi E=0$, the volume equation makes $E=0$.  It lies in the
left kernel of $L_\chi(k_*)$, since
$\ell_E L_\chi=\langle\chi E,\cdot\rangle-
\langle K(k_*)\chi E,\chi\,\cdot\rangle=0$.
It is a nonzero functional also on the chosen sector, because that
sector is closed under complex conjugation and multiplication by
the real rotationally invariant $\chi$.  Its derivative pairing
$\ell_E L_\chi'(k_*)E$ is exactly $d$.
The scalar Schur reduction in Lemma~\ref{ms:continuation} shows
that this pairing is nonzero at a simple characteristic value;
otherwise the scalar zero would have multiplicity at least two.
The residue of $L_\chi^{-1}$ is therefore
$E\otimes\ell_E/d$.

The vacuum electric resolvent is $G_0(k)=k^{-2}K(k)$, with its
natural restriction when the source is supported in $B$.
The full electric field for a compact source is
$L_\chi(k)^{-1}G_0(k)f$ in $B$.  At the pole, reciprocity and
$K(k_*)\chi E=E$ give
\begin{equation}
 \ell_E G_0(k_*)f=k_*^{-2}\int E(y)\cdot f(y)\,dy.
\end{equation}
The same identity for sources outside $B$ follows directly from
the global volume representation and the symmetric kernel.
Multiplying the two residue factors yields
$E\otimes E/(k_*^2d)$, which is
\eqref{ms:green-residue}.  This is a genuine pole normalization
of a quasinormal mode, consistent with reciprocal Green-residue
normalizations\cite{muljarov2016}; the sign reflects the displayed
choice of Maxwell operator.  For a degenerate full-space pole the
formula describes the selected symmetry block, not a claim that
the entire Green residue has rank one.

Under rounding, $\chi_j E_j\to\chi E$ in $L^2$: split the
difference into the uniformly bounded multiplier times $E_j-E$
and dominated convergence applied to $(\chi_j-\chi)E$.
Operator-norm continuity of $K'(k)$ proves convergence of $d$
and $\mathcal N$.  Holomorphy and quadratic scaling follow from
the formulas.  A nonzero analytic square root of $\mathcal N$
on a sufficiently small parameter disk normalizes the field.
